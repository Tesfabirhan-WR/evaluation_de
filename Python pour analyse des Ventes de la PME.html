Python pour analyse des Ventes de la PME

This script reads three Excel files, loads them as three tables in a local SQLite database, then lists the tables to confirm the load.

sales.xlsxstores.xlsxproducts.xlsx become sales.db with tables salesstoresproducts

Imports

import sqlite3
from pathlib import Path
import pandas as pd

DATA = Path(__file__).parent / "data"

Chargement des fichiers Excel

def data_loader():
    path = r"data\sales.db"
    df = []
    files = ["sales", "stores", "products"]
    for file in files:
        df.append(pd.read_excel(rf"data\{file}.xlsx"))

    return df

df_sales, df_stores, df_products = data_loader()

Écriture dans SQLite

conn = sqlite3.connect(Path(__file__).parent / "sales.db")
try:
    df_sales.to_sql("sales", conn, if_exists="replace", index=False)
    df_products.to_sql("products", conn, if_exists="replace", index=False)
    df_stores.to_sql("stores", conn, if_exists="replace", index=False)
finally:
    conn.close()

print("Données insérées avec succès dans SQLite.")
Données insérées avec succès dans SQLite.

Vérification des tables

conn = sqlite3.connect("sales.db")
print(pd.read_sql("SELECT name FROM sqlite_master WHERE type='table'", conn))
conn.close()
